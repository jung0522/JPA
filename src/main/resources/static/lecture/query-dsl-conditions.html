QueryDSL 조건 분리란?

정의

동일한 검색 조건들이 여러 쿼리에서 반복될 때, 공통 조건들을 별도 메서드로 분리하여 재사용할 수 있게 하는 기법입니다.

핵심: 반복되는 조건을 메서드로 분리하여 코드 중복을 제거하고 재사용성 향상

Step 1: 기본 JPQL 쿼리

// CourseRepository.java (Step 1)
@Query("""
    SELECT c FROM Course c
    JOIN FETCH c.instructor
    WHERE c.deleted = false
    AND (:keyword IS NULL OR c.name LIKE %:keyword%)
    AND c.currentStudents < c.maxStudents
    ORDER BY c.createdAt DESC
    """)
Page<Course> searchAvailableCourses(@Param("keyword") String keyword, Pageable pageable);

// 특징:
// - 문자열 기반 JPQL
// - 조건이 쿼리 내부에 하드코딩
// - 동일한 조건을 다른 쿼리에서 재사용하기 어려움

Step 2: QueryDSL + 조건 분리

// CourseQueryRepository.java (Step 2)
// 동일한 조건들이 반복되어 조건 분리 필요!

// 1. 검색 쿼리
public Page<Course> searchCourses(CourseSearchCondition condition, Pageable pageable) {
    return queryFactory
        .selectFrom(course)
        .where(
            courseNameContains(condition.getCourseName()),        // 반복 조건
            instructorNameContains(condition.getInstructorName()) // 반복 조건
        )
        .fetch();
}

// 2. 통계 쿼리에서도 동일한 조건 사용
public List<Course> findCoursesByInstructor(String instructorName) {
    return queryFactory
        .selectFrom(course)
        .where(instructorNameContains(instructorName))  // 동일한 조건 반복!
        .fetch();
}

조건 분리가 필요한 이유

JPQL의 한계

  • • 조건 중복: 동일한 검색 조건이 여러 쿼리에 반복
  • • 재사용 불가: 조건을 다른 쿼리에서 재사용하기 어려움
  • • 유지보수 어려움: 조건 변경 시 모든 쿼리 수정 필요
  • • 타입 안전성 부족: 문자열 기반으로 런타임 오류 가능

QueryDSL 조건 분리의 장점

  • • 조건 재사용: 공통 조건을 메서드로 분리하여 재사용
  • • 유지보수성: 조건 변경 시 한 곳만 수정
  • • 타입 안전성: 컴파일 타임에 오류 검출
  • • 가독성 향상: 쿼리 로직이 명확하게 분리

// 조건들을 조합하여 사용
public Page<Course> searchCourses(SearchCondition condition, Pageable pageable) {
    List<Course> content = queryFactory
        .selectFrom(course)
        .leftJoin(course.instructor, instructor).fetchJoin()
        .where(
            courseNameContains(condition.getName()),
            instructorNameContains(condition.getInstructor()),
            priceBetween(condition.getMinPrice(), condition.getMaxPrice()),
            statusEquals(condition.getStatus()),
            categoryEquals(condition.getCategory())
        )
        .orderBy(course.createdAt.desc())
        .offset(pageable.getOffset())
        .limit(pageable.getPageSize())
        .fetch();

    long total = queryFactory
        .selectFrom(course)
        .where(
            courseNameContains(condition.getName()),
            instructorNameContains(condition.getInstructor()),
            priceBetween(condition.getMinPrice(), condition.getMaxPrice()),
            statusEquals(condition.getStatus()),
            categoryEquals(condition.getCategory())
        )
        .fetch()
        .size();

    return new PageImpl<>(content, pageable, total);
}

2. 범위 조건 패턴

// 범위 조건들
public BooleanExpression priceBetween(BigDecimal minPrice, BigDecimal maxPrice) {
    BooleanExpression min = minPrice != null ? 
        course.price.goe(minPrice) : null;
    BooleanExpression max = maxPrice != null ? 
        course.price.loe(maxPrice) : null;
    return min != null && max != null ? min.and(max) : 
           min != null ? min : max;
}

public BooleanExpression dateBetween(LocalDate startDate, LocalDate endDate) {
    BooleanExpression start = startDate != null ? 
        course.startDate.goe(startDate) : null;
    BooleanExpression end = endDate != null ? 
        course.endDate.loe(endDate) : null;
    return start != null && end != null ? start.and(end) : 
           start != null ? start : end;
}

public BooleanExpression studentCountBetween(Integer minStudents, Integer maxStudents) {
    BooleanExpression min = minStudents != null ? 
        course.currentStudents.goe(minStudents) : null;
    BooleanExpression max = maxStudents != null ? 
        course.currentStudents.loe(maxStudents) : null;
    return min != null && max != null ? min.and(max) : 
           min != null ? min : max;
}

3. 복합 조건 패턴

// 다중 선택 조건
public BooleanExpression statusIn(List<CourseStatus> statuses) {
    return statuses != null && !statuses.isEmpty() ? 
        course.status.in(statuses) : null;
}

public BooleanExpression categoryIn(List<String> categories) {
    return categories != null && !categories.isEmpty() ? 
        course.category.in(categories) : null;
}

// 날짜 범위 조건
public BooleanExpression createdAfter(LocalDateTime dateTime) {
    return dateTime != null ? 
        course.createdAt.goe(dateTime) : null;
}

public BooleanExpression createdBefore(LocalDateTime dateTime) {
    return dateTime != null ? 
        course.createdAt.loe(dateTime) : null;
}
// 중첩 조건
public BooleanExpression isAvailable() {
    return course.currentStudents.lt(course.maxStudents)
           .and(course.status.eq(CourseStatus.ACTIVE));
}

public BooleanExpression isPopular(int minEnrollments) {
    return course.enrollments.size().goe(minEnrollments);
}

// 조건 조합
public BooleanExpression searchByText(String searchText) {
    if (searchText == null || searchText.trim().isEmpty()) {
        return null;
    }
    
    return course.name.containsIgnoreCase(searchText)
           .or(course.description.containsIgnoreCase(searchText))
           .or(course.instructor.name.containsIgnoreCase(searchText));
}

@Repository
@RequiredArgsConstructor
public class CourseQueryRepository {
    
    private final JPAQueryFactory queryFactory;
    private final QCourse course = QCourse.course;
    private final QUser instructor = QUser.user;
    
    // 기본 검색 조건들
    public BooleanExpression courseNameContains(String name) {
        return name != null ? 
            course.name.containsIgnoreCase(name) : null;
    }
    
    public BooleanExpression instructorNameContains(String instructorName) {
        return instructorName != null ? 
            course.instructor.fullName.containsIgnoreCase(instructorName) : null;
    }
    
    public BooleanExpression priceBetween(BigDecimal minPrice, BigDecimal maxPrice) {
        BooleanExpression min = minPrice != null ? 
            course.price.goe(minPrice) : null;
        BooleanExpression max = maxPrice != null ? 
            course.price.loe(maxPrice) : null;
        return min != null && max != null ? min.and(max) : 
               min != null ? min : max;
    }
    
    public BooleanExpression statusEquals(CourseStatus status) {
        return status != null ? course.status.eq(status) : null;
    }
    
    public BooleanExpression isAvailable() {
        return course.currentStudents.lt(course.maxStudents)
               .and(course.status.eq(CourseStatus.ACTIVE));
    }
    
    // 검색 메서드
    public Page<Course> searchCourses(CourseSearchCondition condition, Pageable pageable) {
        List<Course> content = queryFactory
            .selectFrom(course)
            .leftJoin(course.instructor, instructor).fetchJoin()
            .where(
                courseNameContains(condition.getName()),
                instructorNameContains(condition.getInstructor()),
                priceBetween(condition.getMinPrice(), condition.getMaxPrice()),
                statusEquals(condition.getStatus()),
                isAvailable()
            )
            .orderBy(course.createdAt.desc())
            .offset(pageable.getOffset())
            .limit(pageable.getPageSize())
            .fetch();

        long total = queryFactory
            .selectFrom(course)
            .where(
                courseNameContains(condition.getName()),
                instructorNameContains(condition.getInstructor()),
                priceBetween(condition.getMinPrice(), condition.getMaxPrice()),
                statusEquals(condition.getStatus()),

            )
            .fetch()
            .size();

        return new PageImpl<>(content, pageable, total);
    }
}

사용 예시

// Service에서 사용
@Service
@RequiredArgsConstructor
public class CourseService {
    
    private final CourseQueryRepository courseQueryRepository;
    
    public Page<CourseResponse> searchCourses(CourseSearchCondition condition, Pageable pageable) {
        Page<Course> courses = courseQueryRepository.searchCourses(condition, pageable);
        return courses.map(CourseResponse::from);
    }
    
    // 특정 조건만으로 검색
    public List<CourseResponse> findAvailableCourses() {
        List<Course> courses = courseQueryRepository.queryFactory
            .selectFrom(courseQueryRepository.course)
            .where(courseQueryRepository.isAvailable())
            .orderBy(courseQueryRepository.course.createdAt.desc())
            .fetch();
        return courses.stream()
            .map(CourseResponse::from)
            .collect(Collectors.toList());
    }
    
    // 강사별 강의 검색
    public List<CourseResponse> findCoursesByInstructor(String instructorName) {
        List<Course> courses = courseQueryRepository.queryFactory
            .selectFrom(courseQueryRepository.course)
            .where(courseQueryRepository.instructorNameContains(instructorName))
            .orderBy(courseQueryRepository.course.createdAt.desc())
            .fetch();
        return courses.stream()
            .map(CourseResponse::from)
            .collect(Collectors.toList());
    }
}
